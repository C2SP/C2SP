# This webtest script is executed against a test C2SP repository imported from
# testrepo_export.txt.
#
# Any :NNN marks (like ":1") are replaced with the corresponding SHA-1 hash,
# while :NNN! is replaced with the SHA-1 hash truncated to 16 characters.

# foo: has releases, @latest redirects to the highest non-prerelease (v2.0.0)
# so that the resulting link can be copied as a stable reference

GET https://c2sp.org/foo
redirect == /foo@v2.0.0

GET https://c2sp.org/foo@latest
code == 200
body contains <a class="current" href="/foo@v2.0.0">v2.0.0</a>

# foo: explicit versions, including legacy ones without front matter

GET https://c2sp.org/foo@v2.0.0
code == 200
header content-type == text/html; charset=utf-8
body contains <h1>The Foo Specification</h1>
body contains c2sp.org/foo
body contains <a class="current" href="/foo@v2.0.0">v2.0.0</a>
body contains href="/foo@v1.0.0"
body contains href="/foo@main"
body contains https://github.com/foomaint
body contains @foomaint2
body contains https://github.com/C2SP/C2SP/issues/new?title=foo%3A%20
body contains https://github.com/C2SP/C2SP/blob/foo/v2.0.0/foo.md
body contains <h2 id="the-keyed_hash-section">
body contains <a class="anchor" href="#the-keyed_hash-section"
body contains <h2 id="dup">
body contains <h2 id="dup-1">
body contains <blockquote class="alert alert-note">
body contains <blockquote>
body !contains foo-alias.example
body !contains [!WARNING]
body !contains [!NOTE]
body !contains editor&#39;s copy
body !contains description: Test spec foo
body !contains class="notice"

# Accept: text/markdown serves the source, without front matter and the
# warning boilerplate

GET https://c2sp.org/foo@v2.0.0
reqheader Accept text/markdown
code == 200
header content-type == text/markdown; charset=utf-8
header vary == Accept
body contains # The Foo Specification
body !contains description: Test spec foo
body !contains [!WARNING]
body !contains <h1>

GET https://c2sp.org/foo@v1.0.0
reqheader Accept text/markdown
code == 200
body contains foo content 2

GET https://c2sp.org/-/coc
reqheader Accept text/markdown
code == 200
header content-type == text/markdown; charset=utf-8
body contains # Test Code of Conduct

GET https://c2sp.org/foo@v0.0.1
code == 200
body contains foo content 1
body contains href="/foo@v2.0.0"

GET https://c2sp.org/foo@v1.0.0
code == 200
body contains foo content 2

GET https://c2sp.org/foo@v1.0.1-pre.1
code == 200
body contains foo content 3

# foo: @<commit-hash> reachable from origin/main

GET https://c2sp.org/foo@:18
code == 200
body contains historical snapshot
body contains <code>:18</code>
body contains href="/foo@v2.0.0">v2.0.0</a>
body contains foo content 4
body contains https://github.com/C2SP/C2SP/blob/:18/foo.md

GET https://c2sp.org/foo@:18!
code == 200
body contains foo content 4

# foo: @main

GET https://c2sp.org/foo@main
code == 200

body contains href="/foo@v2.0.0">v2.0.0</a>
body contains <h1>The Foo Specification</h1>
body contains https://github.com/C2SP/C2SP/blob/main/foo.md

# foo: non-existent version

code == 404

# commit hash not reachable from origin/main

GET https://c2sp.org/foo@:20

# invalid ref that looks like it could be a version or hash

GET https://c2sp.org/foo@notahex
code == 404
